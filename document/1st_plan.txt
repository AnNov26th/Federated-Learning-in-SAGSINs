assets/
└── textures/
    ├── earth_day.jpg

    ├── earth_normal.jpg

Lúc đó earth.js sẽ chuyển từ:

const material = new THREE.MeshStandardMaterial({
    roughness: 0.8,
    metalness: 0.0
});

sang kiểu:

const textureLoader = new THREE.TextureLoader();

const earthTexture =
    textureLoader.load(
        "/assets/textures/earth_day.jpg"
    );

const material =
    new THREE.MeshStandardMaterial({
        map: earthTexture
    });
